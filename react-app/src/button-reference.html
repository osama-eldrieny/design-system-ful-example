React Implementation

import React from 'react';
import './Button.css';

// Button component props interface with all variants
interface ButtonProps
  extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  /** Button variant: Primary or Secondary */
  variant?: 'primary' | 'secondary';

  /** Button style: Filled or Text */
  style?: 'filled' | 'text';

  /** Button size: Small, Medium, or Icon */
  size?: 'small' | 'medium' | 'icon';

  /** Button state: Default, Hover, Focus, Disabled */
  state?: 'default' | 'hover' | 'focus' | 'disabled';

  /** Button content/label */
  children: React.ReactNode;
}

/**
 * Button Component
 * A versatile button with multiple variants (Primary/Secondary),
 * styles (Filled/Text), sizes (Small/Medium/Icon), and states.
 */
export const Button = React.forwardRef<
  HTMLButtonElement,
  ButtonProps
>(
  (
    {
      variant = 'primary',
      style = 'filled',
      size = 'medium',
      state = 'default',
      children,
      className,
      ...props
    },
    ref
  ) => {
    // Build base className from variant, style, state
    const baseClass = `btn btn--${variant}-${style}-${state}`;

    // Add size modifier if not medium (medium is default)
    const sizeClass =
      size !== 'medium' ? ` btn--${size}` : '';

    // Combine classes and filter empty strings
    const finalClassName = [
      baseClass,
      sizeClass,
      className,
    ]
      .filter(Boolean)
      .join(' ');

    return (
      <button
        ref={ref}
        className={finalClassName}
        disabled={
          state === 'disabled' || props.disabled
        }
        {...props}
      >
        {size === 'icon' ? '+' : children}
      </button>
    );
  }
);

Button.displayName = 'Button';

Usage Examples

// Primary Filled (Default)
<Button>Click Me</Button>
<Button variant="primary" style="filled" size="medium">
  Submit
</Button>

// Secondary Text Button
<Button variant="secondary" style="text">Cancel</Button>

// Small Size Button
<Button size="small">Small</Button>

// Icon Only Button
<Button size="icon" aria-label="Add item" />

// Disabled State
<Button state="disabled">Disabled</Button>
