Input Field

A text input component for form data entry. Provides visual feedback for different interaction states: Default, Hover, and Focus. Supports multi-theme variants with customizable padding, border radius, and typography.

React Implementation

interface InputFieldProps {
  placeholder?: string;
  value?: string;
  onChange?: (e: React.ChangeEvent<HTMLInputElement>) => void;
  onFocus?: () => void;
  onBlur?: () => void;
  disabled?: boolean;
  className?: string;
}

export const InputField = ({
  placeholder = 'Enter text here',
  value = '',
  onChange,
  onFocus,
  onBlur,
  disabled = false,
  className = "",
}: InputFieldProps) => {
  return (
    <input
      type="text"
      className={`input-field ${className}`}
      placeholder={placeholder}
      value={value}
      onChange={onChange}
      onFocus={onFocus}
      onBlur={onBlur}
      disabled={disabled}
    />
  );
};
