Radio Button

A simple and accessible radio button component for selecting a single option from a group. Supports both selected (active) and unselected (default) states with visual feedback.

React Implementation

import React from 'react';
import './RadioButton.css';

interface RadioButtonProps extends React.InputHTMLAttributes<HTMLInputElement> {
  label?: string;
  disabled?: boolean;
}

export const RadioButton = React.forwardRef<
  HTMLInputElement,
  RadioButtonProps
>(
  ({ label, disabled, ...props }, ref) => {
    return (
      <>
        <input
          ref={ref}
          type="radio"
          disabled={disabled}
          className="radio-button"
          {...props}
        />
        {label && <label>{label}</label>}
      </>
    );
  }
);

RadioButton.displayName = 'RadioButton';

Usage Examples

// Basic Radio Button
<RadioButton name="option" value="a" />

// With Label
<RadioButton
  name="option"
  value="a"
  label="Option A"
/>

// Checked and Disabled
<RadioButton
  name="option"
  value="b"
  label="Option B"
  checked={true}
  disabled={false}
/>
