Avatar

A circular or rounded image component used to represent users and profiles. Supports multiple sizes and border radius modes for flexible visual presentation.

React Implementation

import React from 'react';
import '../assets/tokens.css';

interface AvatarProps extends React.ImgHTMLAttributes<HTMLImageElement> {
  size?: 'small' | 'xlarge';
  radius?: 'square' | 'round' | 'pills';
  src: string;
  alt: string;
}

export const Avatar = React.forwardRef<
  HTMLDivElement,
  AvatarProps
>(
  (
    {
      size = 'xlarge',
      radius = 'pills',
      src,
      alt,
      className,
      ...props
    },
    ref
  ) => {
    const classes = [
      'avatar',
      `avatar--${size}`,
      `avatar--${radius}`,
      className
    ].filter(Boolean).join(' ');

    return (
      <div ref={ref} className={classes}>
        <img src={src} alt={alt} {...props} />
      </div>
    );
  }
);

Avatar.displayName = 'Avatar';

Usage Examples

// Default Avatar (XLarge, Pills)
<Avatar src="/avatar.jpg" alt="User" />

// Small Avatar
<Avatar size="small" src="/avatar.jpg" alt="User" />

// Square border radius
<Avatar radius="square" src="/avatar.jpg" alt="User" />

// Round border radius
<Avatar radius="round" src="/avatar.jpg" alt="User" />

// Avatar Group
<div className="avatar-group">
  <Avatar size="small" src="/user1.jpg" alt="User 1" />
  <Avatar size="small" src="/user2.jpg" alt="User 2" />
  <Avatar size="small" src="/user3.jpg" alt="User 3" />
</div>
