Pagination

Pagination component for navigating through pages. Displays page numbers, previous/next buttons, and supports active/default/hover states. Uses nested Pagination number and Pagination arrow components.

React Implementation

import React, { useState } from 'react';
import './Pagination.css';

interface PaginationProps {
  totalPages: number;
  currentPage: number;
  onPageChange: (page: number) => void;
  showPrevious?: boolean;
  showNext?: boolean;
}

export const Pagination = React.forwardRef<
  HTMLDivElement,
  PaginationProps
>(({ totalPages, currentPage, onPageChange, showPrevious = true, showNext = true }, ref) => {
  const pages = Array.from({ length: totalPages }, (_, i) => i + 1);

  return (
    <div ref={ref} className="pagination-container">
      {showPrevious && (
        <button
          className="pagination-item pagination-arrow"
          onClick={() => onPageChange(currentPage - 1)}
          disabled={currentPage === 1}
        >
          <i className="fa-solid fa-chevron-left"></i>
        </button>
      )}
      {pages.map((page) => (
        <button
          key={page}
          className={`pagination-item pagination-number ${ currentPage === page ? 'pagination-item--active' : '' }`}
          onClick={() => onPageChange(page)}
        >
          {page}
        </button>
      ))}
      {showNext && (
        <button
          className="pagination-item pagination-arrow"
          onClick={() => onPageChange(currentPage + 1)}
          disabled={currentPage === totalPages}
        >
          <i className="fa-solid fa-chevron-right"></i>
        </button>
      )}
    </div>
  );
});

Pagination.displayName = 'Pagination';

Usage Examples

// Basic Pagination with 5 pages
<Pagination
  totalPages={5}
  currentPage={currentPage}
  onPageChange={(page) => setCurrentPage(page)}
/>

// Pagination without arrows
<Pagination
  totalPages={10}
  currentPage={currentPage}
  onPageChange={(page) => setCurrentPage(page)}
  showPrevious={false}
  showNext={false}
/>

// Pagination with disabled arrows
<Pagination
  totalPages={5}
  currentPage={currentPage}
  onPageChange={(page) => setCurrentPage(page)}
  showPrevious={currentPage > 1}
  showNext={currentPage < 5}
/>
