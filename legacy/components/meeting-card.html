Meeting Card

A card component for displaying meeting information with title, time, and attendee avatars. Available in three color variants (Primary, Success, Danger) for status indication. Contains nested Avatar Group component with up to 5 attendee avatars.

React Implementation

import React from 'react';
import './MeetingCard.css';

interface Avatar {
  initials: string;
  name: string;
}

interface MeetingCardProps extends React.HTMLAttributes<HTMLDivElement> {
  variant?: 'primary' | 'success' | 'danger';
  title: string;
  time: string;
  avatars: Avatar[];
}

export const MeetingCard = React.forwardRef<
  HTMLDivElement,
  MeetingCardProps
>(({ variant = 'primary', title, time, avatars, className, ...props }, ref) => (
  <div
    ref={ref}
    className={`meeting-card meeting-card--${variant} ${className || ''}`}
    {...props}
  >
    <div className="meeting-card-info">
      <h3 className="meeting-card-title">{title}</h3>
      <p className="meeting-card-time">{time}</p>
    </div>
    <AvatarGroup avatars={avatars} />
  </div>
));

MeetingCard.displayName = 'MeetingCard';
