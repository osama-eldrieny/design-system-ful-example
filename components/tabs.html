Tabs

Tab navigation component for organizing content into separate views. Contains multiple tab items with support for default, hover, and active states. Supports icons and labels for each tab.

React Implementation

import React, { useState } from 'react';
import './Tabs.css';

interface TabItem {
  id: string;
  label: string;
  icon?: string;
  content?: React.ReactNode;
}

interface TabsProps {
  items: TabItem[];
  defaultActiveId?: string;
  showIcons?: boolean;
  onTabChange?: (id: string) => void;
}

export const Tabs = React.forwardRef<
  HTMLDivElement,
  TabsProps
>(({ items, defaultActiveId, showIcons = true, onTabChange }, ref) => {
  const [activeId, setActiveId] = useState(defaultActiveId || items[0]?.id);

  const handleTabClick = (id: string) => {
    setActiveId(id);
    onTabChange?.(id);
  };

  return (
    <div ref={ref} className="tabs-container">
      {items.map((item) => (
        <button
          key={item.id}
          className={`tab-item ${activeId === item.id ? 'tab-item--active' : ''}`}
          onClick={() => handleTabClick(item.id)}
        >
          {showIcons && item.icon && (
            <i className={`tab-item-icon ${item.icon}`} />
          )}
          <span>{item.label}</span>
        </button>
      ))}
    </div>
  );
});

Tabs.displayName = 'Tabs';

Usage Examples

// Basic Tabs with 3 items
const tabs = [
  { id: 'home', label: 'Home', icon: 'fa-solid fa-home' },
  { id: 'profile', label: 'Profile', icon: 'fa-solid fa-user' },
  { id: 'messages', label: 'Messages', icon: 'fa-solid fa-envelope' }
];

<Tabs items={tabs} />

// Without icons
<Tabs
  items={tabs}
  showIcons={false}
/>

// With default active tab and change handler
<Tabs
  items={tabs}
  defaultActiveId="profile"
  onTabChange={(id) => console.log(id)}
/>
