Footer

React Implementation

import './footer.css';

interface FooterProps {
  className?: string;
}

export const Footer = ({
  className = "",
}: FooterProps) => {
  return (
    <footer className={`footer ${className}`}>
      <div className="footer-text">
        {/* Content from --footer-title-content */}
        content("var(--footer-title-content)")
      </div>
      <div className="footer-links">
        <button>
          {/* --footer-link1-content */}
          content("var(--footer-link1-content)")
        </button>
        <button>
          content("var(--footer-link2-content)")
        </button>
        <button>
          content("var(--footer-link3-content)")
        </button>
      </div>
    </footer>
  );
};

Usage Examples

// Basic Footer
<Footer />

// With Custom Copyright Text
<Footer
  copyrightText="© 2025 My Company. All rights reserved."
/>

// With Links
<Footer
  links={[
    { label: "License", url: "/license" },
    { label: "Help", url: "/help" },
    { label: "Policy", url: "/policy" },
  ]}
/>
