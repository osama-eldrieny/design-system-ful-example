React Implementation

import React, { useState } from 'react';

interface NotificationListItemProps {
  appName: string;
  appIcon: string;
  isEnabled: boolean;
  onToggle: (enabled: boolean) => void;
}

export function NotificationListItem({
  appName,
  appIcon,
  isEnabled,
  onToggle
}: NotificationListItemProps) {
  return (
    <div className="notification-list-item">
      <div className="app-list-item">
        <div className="app-icon">{appIcon}</div>
        <span className="app-title">{appName}</span>
      </div>
      <button
        className={`toggle-switch ${isEnabled ? 'active' : 'inactive'}`}
        onClick={() => onToggle(!isEnabled)}
        aria-label={`Toggle ${appName}`}
      >
        <div className="toggle-circle" />
      </button>
    </div>
  );
}

Usage Examples

// Default notification item
<NotificationListItem
  appName="Google"
  appIcon="📱"
  isEnabled={true}
  onToggle={(enabled) => console.log(enabled)}
/>

// Disabled notification item
<NotificationListItem
  appName="X (Twitter)"
  appIcon="𝕏"
  isEnabled={false}
  onToggle={(enabled) => console.log(enabled)}
/>

// With state management
const [enabled, setEnabled] = useState(true);
<NotificationListItem
  appName="LinkedIn"
  appIcon="in"
  isEnabled={enabled}
  onToggle={setEnabled}
/>
