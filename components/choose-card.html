Choose Card

React Implementation

import React from 'react';
import './ChooseCard.css';

interface ChooseCardProps extends React.HTMLAttributes<HTMLDivElement> {
  title: string;
  description: string;
  price?: string;
  selected?: boolean;
  onChange?: (selected: boolean) => void;
}

export const ChooseCard = React.forwardRef<
  HTMLDivElement,
  ChooseCardProps
>(
  ({
    title,
    description,
    price,
    selected = false,
    onChange,
    className,
    ...props
  },
  ref
>) => {
  return (
    <div
      ref={ref}
      className={`choose-card ${selected ? 'selected' : ''} ${className}`}
      onClick={() => onChange?.(!selected)}
      {...props}
    >
      <input type="radio" checked={selected} readOnly />
      <div className="choose-card-content">
        <div className="choose-card-title">{title}</div>
        <div className="choose-card-desc">{description}</div>
      </div>
      {price && <div className="choose-card-price">{price}</div>}
    </div>
  );
}
);

ChooseCard.displayName = 'ChooseCard';

Usage Examples

// Basic Choose Card
<ChooseCard
  title="Pro Plan"
  description="Advanced tools for growing professional teams."
  price="Free"
/>

// With selection handler
<ChooseCard
  title="Enterprise"
  description="Custom solutions for large organizations."
  price="$299/mo"
  onChange={(selected) => console.log(selected)}
/>

// Pre-selected
<ChooseCard
  title="Basic"
  description="Perfect for individuals and small teams."
  price="$49/mo"
  selected={true}
/>
