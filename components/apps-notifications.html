React Implementation

import React, { useState } from 'react';

interface App {
  id: string;
  name: string;
  icon: string;
  isEnabled: boolean;
}

interface AppsNotificationsProps {
  apps: App[];
  onToggle: (appId: string, enabled: boolean) => void;
  brand?: 'diamond' | 'amber' | 'opal';
  showIcons?: boolean;
}

export function AppsNotifications({
  apps,
  onToggle,
  brand = 'diamond',
  showIcons = true
}: AppsNotificationsProps) {
  return (
    <div className={`apps-notifications apps-notifications-${brand}`}>
      <h3 className="apps-notifications-title">
        Apps Notification
      </h3>
      <div className="apps-list">
        {apps.map((app) => (
          <div key={app.id} className="notification-list-item">
            <div className="app-list-item">
              {showIcons && (
                <img src={app.icon} alt={app.name} className="app-icon" />
              )}
              <span className="app-title">{app.name}</span>
            </div>
            <button
              className={`toggle-switch ${app.isEnabled ? 'active' : 'inactive'}`}
              onClick={() => onToggle(app.id, !app.isEnabled)}
              aria-label={`Toggle ${app.name} notifications`}
            >
              <div className="toggle-circle" />
            </button>
          </div>
        ))}
      </div>
    </div>
  );
}

Accessibility & Notes

Best Practices

  • Always provide meaningful app names and icons for clear identification
  • Toggle states should be clearly distinguishable with sufficient contrast
  • App icons should be 16px × 16px with proper aspect ratio
  • Ensure toggle switches are keyboard accessible (Tab, Space/Enter to toggle)
  • Consider showing icons conditionally based on space availability
  • Use proper ARIA labels for screen reader users
  • Maintain consistent spacing and alignment across all items
  • Respect user's density preference (Comfortable vs Compact)
  • Support multiple language content through Component Language collection
  • Support theme variations through Component Brand collection (Diamond, Amber, Opal)
