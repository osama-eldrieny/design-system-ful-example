React Implementation

import React from 'react';
import './Navbar.css';
import { Logo } from './Logo';
import { MenuItem } from './MenuItem';

interface NavbarProps extends React.HTMLAttributes<HTMLElement> {
  logo?: React.ReactNode;
  items?: Array<{ label: string; icon?: string; onClick?: () => void }>;
}

export const Navbar = React.forwardRef<
  HTMLElement,
  NavbarProps
>(({ logo, items = [], className, ...props }, ref) => {
  return (
    <nav ref={ref} className={`navbar-preview ${className || ''}`} {...props}>
      <div className="navbar-logo">
        {logo || <Logo />}
      </div>
      <div className="navbar-items">
        {items.map((item) => (
          <MenuItem
            key={item.label}
            label={item.label}
            icon={item.icon}
            onClick={item.onClick}
          />
        ))}
      </div>
    </nav>
  );
});

Navbar.displayName = 'Navbar';

Usage Examples

// Basic Navbar with default Logo
<Navbar
  items={[
    { label: "Home", onClick: () => {} },
    { label: "Themes", onClick: () => {} },
    { label: "More", icon: "chevron-down", onClick: () => {} }
  ]}
/>

// Navbar with custom Logo
<Navbar
  logo={<CustomLogo />}
  items={[...]}
/>

// With custom className
<Navbar
  className="navbar-custom"
  items={[...]}
/>
