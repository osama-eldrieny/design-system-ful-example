React Implementation

import React, { useState } from 'react';
import './Dropdown.css';

interface DropdownItem {
  id: string;
  label: string;
  icon: string;
  onClick?: () => void;
}

interface DropdownProps {
  items: DropdownItem[];
  onItemClick?: (itemId: string) => void;
}

export const Dropdown = React.forwardRef<
  HTMLDivElement,
  DropdownProps
>(({ items, onItemClick }, ref) => {
  const [hoverIndex, setHoverIndex] = useState<number | null>(null);

  return (
    <div ref={ref} className="dropdown">
      {items.map((item, idx) => (
        <div
          key={item.id}
          className={`dropdown__item ${
            hoverIndex === idx ? 'dropdown__item--hover' : ''
          }`}
          onMouseEnter={() => setHoverIndex(idx)}
          onMouseLeave={() => setHoverIndex(null)}
          onClick={() => {
            item.onClick?.();
            onItemClick?.(item.id);
          }}
        >
          <i className={item.icon}></i>
          <span>{item.label}</span>
        </div>
      ))}
    </div>
  );
});

Dropdown.displayName = 'Dropdown';

Usage Examples

// Basic Dropdown with 3 items
const items = [
  { id: 'edit', label: 'Edit', icon: 'fa-solid fa-pen' },
  { id: 'dup', label: 'Duplicate', icon: 'fa-solid fa-copy' },
  { id: 'del', label: 'Delete', icon: 'fa-solid fa-trash' }
];

<Dropdown
  items={items}
  onItemClick={(id) => console.log(id)}
/>

// With custom click handlers
<Dropdown
  items={[
    {
      id: 'edit',
      label: 'Edit',
      icon: 'fa-solid fa-pen',
      onClick: () => handleEdit()
    }
  ]}
/>

Usage Guidelines

Best Practices

  • ✓ Keep menu items concise and action-oriented
  • ✓ Use descriptive labels that clearly indicate the action
  • ✓ Limit menu items to 3-7 items for optimal usability
  • ✓ Use icons that are consistent with Font Awesome library
  • ✓ Provide visual feedback on hover state
  • ✓ Position dropdown to avoid viewport overflow

Accessibility

  • ✓ Ensure keyboard navigation support (arrow keys, Enter)
  • ✓ Use semantic HTML elements
  • ✓ Maintain sufficient color contrast (4.5:1 minimum)
  • ✓ Provide focus indicators for keyboard navigation
  • ✓ Support screen reader announcements
