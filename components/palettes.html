Palettes

Color palette component that displays a set of color swatches across different shades (100-900) for Primary, Secondary, Success, Warning, and Danger variants. Each palette contains 9 color shades with proper contrast for text content.

React Implementation

import React from 'react';
import './Palettes.css';

interface PaletteColor {
  shade: number;
  bgColor: string;
  fontColor: string;
}

interface PalettesProps {
  variant: 'primary' | 'secondary' | 'success' | 'warning' | 'danger';
  showShades?: 'all' | 'light' | 'core' | 'dark';
}

export const Palettes = React.forwardRef<
  HTMLDivElement,
  PalettesProps
>(({ variant = 'primary', showShades = 'all' }, ref) => {
  const shadeRanges = {
    all: [100, 200, 300, 400, 500, 600, 700, 800, 900],
    light: [100, 200, 300, 400],
    core: [400, 500, 600],
    dark: [600, 700, 800, 900]
  };

  const shades = shadeRanges[showShades];

  return (
    <div ref={ref} className="palettes-container">
      {shades.map((shade) => (
        <button
          key={shade}
          className={`palette-box palette-${variant}-${shade}`}
          title={`${variant.charAt(0).toUpperCase() + variant.slice(1)} ${shade}`}
        />
      ))}
    </div>
  );
});

Palettes.displayName = 'Palettes';

Usage Examples

// Primary palette with all shades
<Palettes variant="primary" />

// Secondary palette, light shades only
<Palettes
  variant="secondary"
  showShades="light"
/>

// Success palette, core shades
<Palettes
  variant="success"
  showShades="core"
/>

// Warning palette, dark shades
<Palettes
  variant="warning"
  showShades="dark"
/>

// Danger palette
<Palettes variant="danger" />
