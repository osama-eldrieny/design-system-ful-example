Toggle

A toggle switch component for enabling or disabling options. Provides visual feedback for two states: Off and On. Supports smooth transitions with customizable colors, sizing, and border radius.

.toggle-switch {

React Implementation

interface ToggleProps {
  checked?: boolean;
  onChange?: (checked: boolean) => void;
  disabled?: boolean;
  className?: string;
  label?: string;
}

export const Toggle = ({
  checked = false,
  onChange,
  disabled = false,
  className = "",
  label,
}: ToggleProps) => {
  const handleClick = () => {
    if (!disabled && onChange) {
      onChange(!checked);
    }
  };

  return (
    <div className="toggle-container">
      <div
        className={`toggle-switch ${checked ? 'active' : ''} ${className}`}
        onClick={handleClick}
        role="switch"
        aria-checked={checked}
        aria-disabled={disabled}
      >
        <div className="toggle-circle"></div>
      </div>
      {label && <span>{label}</span>}
    </div>
  );
};
